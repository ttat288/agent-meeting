function checkRuntime(adapter) {

function contextConfig(file) {
  if (!file) return { config: { mcpServers: {} }, tools: [], readOnlyPaths: [] };

  const stat = fs.lstatSync(file);
  if (!stat.isFile() || stat.isSymbolicLink() || stat.size > 512 * 1024) throw new Error('Invalid task context file');

  if (!obj || typeof obj !== 'object' || Array.isArray(obj)) return { valid: false, reportedModel: 'unknown', text: '', usage: null, error: 'Runtime result must be an object' };
  const text = String(obj.result ?? obj.response ?? obj.text ?? '');

  const hasError = obj.error != null && obj.error !== false && String(obj.error).trim().length > 0;
  const success = !hasError && (adapter === 'agy' ? obj.status === 'SUCCESS' && text.trim().length > 0 : obj.type === 'result' && obj.subtype === 'success' && !obj.is_error && text.trim().length > 0);

  // agy response contracts vary. Only machine fields count; prose self-identification is not evidence.
  const reportedModel = models.length === 1 ? models[0] : models.length > 1 ? 'unknown' : typeof obj.model === 'string' ? obj.model : 'unknown';

function runCli(options) {
  const { adapter, workspace, timeoutMs = 120000, signal } = options;
  const args = buildArgs(adapter, options); let bin = executable(adapter);
  if (!bin) return Promise.resolve({ status: 'unavailable', reportedModel: 'unknown', error: 'Executable not found', exitCode: null });
  if (!Number.isInteger(timeoutMs) || timeoutMs < 100 || timeoutMs > 600000) throw new Error('Invalid timeout');
  if (!workspace || !fs.statSync(workspace).isDirectory()) throw new Error('Workspace required');
  const cwd = fs.realpathSync(workspace), started = Date.now();
  // agy's plan flag is ineffective with slash expansion disabled. Its native
  // file tools can still edit, so even read-only tasks/probes need a real boundary.
  const wrapped = options.write || options.isolate || options.taskContextFile || adapter === 'agy' ? require('./execution-sandbox').wrapExecution(bin, args, { workspace: cwd, write: options.write === true,
    readOnlyPaths: contextConfig(options.taskContextFile).readOnlyPaths }) : null;
  if (wrapped) bin = wrapped.bin;
  return new Promise(resolve => {
    let stdout = '', stderr = '', stdoutBytes = 0, reason = null, finished = false, closed;
    const child = spawn(bin, wrapped ? wrapped.args : args, { cwd, shell: false, detached: options.detached !== false && process.platform !== 'win32', stdio: ['pipe', 'pipe', 'pipe'] });
    child.stdin.on('error', () => {});
    child.stdin.end(adapter === 'claude' ? options.prompt : '');
    const stop = why => {
      if (finished || reason) return;
      reason = why;
      try { process.platform === 'win32' || options.detached === false ? child.kill('SIGTERM') : process.kill(-child.pid, 'SIGTERM'); } catch {}
      killTimer = setTimeout(() => {
        try { process.platform === 'win32' || options.detached === false ? child.kill('SIGKILL') : process.kill(-child.pid, 'SIGKILL'); } catch {}
        killTimer = null;
        if (closed) done(closed.code, closed.error);
      }, 2000);
    };
    let killTimer;
    const timer = setTimeout(() => stop('timeout'), timeoutMs);
    const abort = () => stop('cancelled');
    signal?.addEventListener('abort', abort, { once: true });
    if (signal?.aborted) stop('cancelled');
    // StringDecoder retains partial UTF-8 sequences between pipe chunks.
    child.stdout.setEncoding('utf8'); child.stderr.setEncoding('utf8');
    child.stdout.on('data', text => { const bytes = Buffer.byteLength(text); if (stdoutBytes + bytes > 1024 * 1024) return stop('output-limit'); stdoutBytes += bytes; stdout += text; });
    child.stderr.on('data', text => { stderr = (stderr + text).slice(-8192).replace(/^[\uDC00-\uDFFF]/, ''); });
    const done = (code, err) => {
      if (finished) return;
      // A silent grandchild can outlive the parent and its stdio. Keep the
      // promise and escalation timer alive until the group has been signalled.
      if (reason && killTimer) { closed ||= { code, error: err }; return; }
      finished = true;
      clearTimeout(timer);
      signal?.removeEventListener('abort', abort);
      wrapped?.cleanup();
      const parsed = parseOutput(adapter, stdout);
      resolve({ adapter, requestedModel: options.modelId, ...parsed, exitCode: code,
        status: reason || (err ? 'failed' : code === 0 && parsed.valid && !parsed.permissionDenials ? 'reported' : 'failed'),
        elapsedMs: Date.now() - started, stdout, stderr, error: err?.message || parsed.error, workspace: cwd });
    };
    child.on('error', e => done(null, e)); child.on('close', code => done(code));
  });
}

'use strict';
// Linux filesystem WRITE isolation. Host files remain readable and network access
// remains available for provider APIs. Credentials/settings are always read-only.
const fs = require('node:fs');
const path = require('node:path');
const os = require('node:os');
const { spawnSync } = require('node:child_process');
const BWRAP = '/usr/bin/bwrap';
const WRITABLE_STATE = Object.freeze(['debug', 'projects', 'todos', 'session-env', 'shell-snapshots', 'statsig', 'cache', 'backups']);
function unsafe(message) { const e = new Error(`Execution sandbox: ${message}`); e.code = 'EXECUTION_SANDBOX_UNAVAILABLE'; return e; }
function checkSandbox() {
  if (process.platform !== 'linux') throw unsafe('Linux bubblewrap is required; refusing unsandboxed execution');
  try { fs.accessSync(BWRAP, fs.constants.X_OK); } catch { throw unsafe('bubblewrap is missing; refusing unsandboxed execution'); }
  const result = spawnSync(BWRAP, ['--ro-bind', '/', '/', '--dev', '/dev', '--proc', '/proc', '--unshare-pid', '--die-with-parent', '--', '/bin/true'], { encoding: 'utf8', timeout: 5000, maxBuffer: 8192 });
  if (result.status !== 0 || result.error) throw unsafe('bubblewrap namespaces are unavailable; refusing unsandboxed execution');
  return { available: true, boundary: 'filesystem-write', readIsolation: false, networkIsolation: false };
}
function within(child, parent) { return parent === path.parse(parent).root || child === parent || child.startsWith(parent + path.sep); }
function placeholder(target, source) {
  fs.mkdirSync(path.dirname(target), { recursive: true, mode: 0o700 });
  if (fs.statSync(source).isDirectory()) fs.mkdirSync(target, { mode: 0o700 });
  else fs.closeSync(fs.openSync(target, 'wx', 0o600));
}
function wrapExecution(bin, args, { workspace, write = false, readOnlyPaths = [] } = {}) {
  checkSandbox();
  if (typeof bin !== 'string' || !path.isAbsolute(bin) || !Array.isArray(args) || args.some(a => typeof a !== 'string' || a.includes('\0'))) throw unsafe('structured executable and string arguments required');
  const executable = fs.realpathSync(bin);
  if (!fs.statSync(executable).isFile()) throw unsafe('executable must be a regular file');
  if (typeof workspace !== 'string' || !path.isAbsolute(workspace) || typeof write !== 'boolean') throw unsafe('absolute workspace and boolean write required');
  const real = fs.realpathSync(workspace), home = fs.realpathSync(os.homedir());
  if (!Array.isArray(readOnlyPaths) || readOnlyPaths.length > 2) throw unsafe('bounded read-only context paths required');
  for (const file of readOnlyPaths) {
    if (typeof file !== 'string' || !path.isAbsolute(file) || fs.realpathSync(file) !== file || within(file, real) || within(real, file)
      || !fs.existsSync(file)) throw unsafe('context paths must be canonical and separate from the writable workspace');
  }
  if (!fs.statSync(real).isDirectory() || within(home, real)) throw unsafe('workspace must be a dedicated directory, not the home or filesystem root');
  const claude = path.join(home, '.claude');
  if (within(real, claude) || within(claude, real)) throw unsafe('workspace overlaps runtime settings');
  // Resolve the parent directory before creating scratch so host symlinks cannot
  // redirect scratch or workspace mount sources after the namespace starts.
  const scratch = fs.mkdtempSync(path.join(fs.realpathSync(os.tmpdir()), 'agent-meeting-sandbox-state-'));
  fs.chmodSync(scratch, 0o700);
  let cleaned = false;
  const cleanup = () => { if (!cleaned) { fs.rmSync(scratch, { recursive: true, force: true }); cleaned = true; } };
  try {
    const tmp = path.join(scratch, 'tmp'); fs.mkdirSync(tmp, { mode: 0o700 });
    // Callers spawn detached with piped stdio. Keep descendants in that
    // managed process group: --new-session would escape cancellation signals.
    const wrap = ['--ro-bind', '/', '/', '--dev', '/dev', '--proc', '/proc', '--unshare-pid', '--die-with-parent', '--cap-drop', 'ALL', '--bind', tmp, '/tmp', '--setenv', 'TMPDIR', '/tmp'];
    // /tmp is hidden. Recreate workspace mount parents in the isolated /tmp
    // before mounting the original host workspace at its exact canonical path.
    if (within(real, '/tmp')) {
      let current = '/tmp';
      for (const part of path.relative('/tmp', real).split(path.sep)) { current = path.join(current, part); wrap.push('--dir', current); }
    }
    wrap.push(write ? '--bind' : '--ro-bind', real, real);
    // Private /tmp hides host task context. Restore only these explicit sources,
    // always read-only, including when worker tools can edit their workspace.
    for (const file of readOnlyPaths) {
      if (!within(file, '/tmp')) continue;
      let current = '/tmp';
      const parent = path.dirname(file);
      for (const part of path.relative('/tmp', parent).split(path.sep).filter(Boolean)) { current = path.join(current, part); wrap.push('--dir', current); }
      wrap.push('--ro-bind', file, file);
    }
    if (fs.existsSync(claude)) {
      if (fs.lstatSync(claude).isSymbolicLink() || !fs.statSync(claude).isDirectory()) throw unsafe('Claude state directory must be a real directory');
      const state = path.join(scratch, 'claude'); fs.mkdirSync(state, { mode: 0o700 });
      const preserved = [];
      for (const name of fs.readdirSync(claude)) {
        if (WRITABLE_STATE.includes(name)) continue;
        const source = path.join(claude, name), target = path.join(state, name);
        // Existing auth, settings, plugins and skills remain read-only mounts.
        // Broken symlinks are not needed and cannot be mounted safely.
        if (!fs.existsSync(source)) continue;
        placeholder(target, source); preserved.push([source, path.join(claude, name)]);
      }
      for (const name of WRITABLE_STATE) fs.mkdirSync(path.join(state, name), { mode: 0o700 });
      if (within(claude, '/tmp')) {
        let current = '/tmp';
        for (const part of path.relative('/tmp', claude).split(path.sep)) { current = path.join(current, part); wrap.push('--dir', current); }
      }
      wrap.push('--bind', state, claude);
      for (const [source, target] of preserved) wrap.push('--ro-bind', source, target);
    }
    wrap.push('--chdir', real, '--', executable, ...args);
    return { bin: BWRAP, args: wrap, cleanup, boundary: 'filesystem-write', readIsolation: false, networkIsolation: false };
  } catch (error) { cleanup(); throw error; }
}
module.exports = { wrapExecution, checkSandbox, WRITABLE_STATE };

test('agy SUCCESS with denied_actions is a permission failure, not a completed report', async () => {
  // Shape observed with agy1.4.5 in the read-only audit, no model call here.
  const output = JSON.stringify({ status: 'SUCCESS', response: 'Partial response before the requested tool was denied.', denied_actions: [{ action: 'command', display_name: 'RunCommand' }] });
  const parsed = parseOutput('agy', output);
  assert.equal(parsed.permissionDenials, 1);
  assert.equal(parsed.error, 'Runtime reported permission denials');
  assert.equal(parsed.reportedModel, 'unknown');
  const { runtime } = fakeRuntime(output);
  const result = await runtime.runCli({ adapter: 'agy', modelId: 'test-id', prompt: 'Read-only audit', workspace: __dirname, timeoutMs: 1000 });
  assert.equal(result.status, 'failed'); assert.equal(result.permissionDenials, 1);
  assert.equal(result.error, 'Runtime reported permission denials');
  assert.equal(parseOutput('agy', JSON.stringify({ status: 'SUCCESS', response: 'Complete response', denied_actions: [] })).permissionDenials, 0);
});


test('agy read-only model calls always request a filesystem write boundary', async () => {
  const { runtime, captured } = fakeRuntime('{"status":"SUCCESS","response":"Read-only report"}');
  const result = await runtime.runCli({ adapter: 'agy', modelId: 'test-id', prompt: 'Read', workspace: __dirname, timeoutMs: 1000 });
  assert.equal(captured.bin, '/fake/bwrap');
  assert.equal(captured.boundary.options.write, false);
  assert.equal(captured.boundary.options.workspace, fs.realpathSync(__dirname));
  assert.equal(captured.cleaned, true); assert.equal(result.status, 'reported');
  assert.equal(result.reportedModel, 'unknown');
});
